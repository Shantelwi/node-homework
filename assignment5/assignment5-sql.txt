SELECT orders.order_id, SUM(products.price * line_items.quantity) AS total_price
FROM orders
JOIN line_items ON orders.order_id = line_items.order_id
JOIN products ON products.product_id = line_items.product_id
GROUP BY orders.order_id
ORDER BY orders.order_id
LIMIT 5;

#Task 2
SELECT customers.customer_name, AVG(t.total_price) AS average_order_price
FROM (
    SELECT orders.customer_id, orders.order_id, SUM(products.price * line_items.quantity) AS total_price
    FROM orders
    JOIN line_items ON orders.order_id = line_items.order_id
    JOIN products ON products.product_id = line_items.product_id
    GROUP BY orders.order_id, orders.customer_id
) AS t
JOIN customers ON customers.customer_id = t.customer_id
GROUP BY customers.customer_id, customers.customer_name
ORDER BY customers.customer_name;

SELECT employees.first_name, employees.last_name, COUNT(orders.order_id) AS order_count
FROM orders 
JOIN employees ON orders.employee_id = employees.employee_id
GROUP BY employees.employee_id, first_name, last_name
HAVING COUNT(orders.order_id) > 5
